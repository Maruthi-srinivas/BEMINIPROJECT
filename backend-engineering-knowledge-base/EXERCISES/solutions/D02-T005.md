## Exercise 1

```python
class Stack:
    def __init__(self):
        self._a: list[int] = []

    def push(self, x: int) -> None:
        self._a.append(x)

    def pop(self) -> int:
        if not self._a:
            raise IndexError("empty")
        return self._a.pop()


def balanced_parens(s: str) -> bool:
    st: list[str] = []
    pairs = {")": "(", "]": "[", "}": "{"}
    for ch in s:
        if ch in "([{":
            st.append(ch)
        elif ch in ")]}":
            if not st or st.pop() != pairs[ch]:
                return False
    return not st
```

Stack ops amortized O(1). `balanced_parens` O(n) time, O(n) space worst case.
