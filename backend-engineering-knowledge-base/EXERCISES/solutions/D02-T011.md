```python
from collections import deque


def bfs_dist(adj: list[list[int]], src: int) -> list[int]:
    n = len(adj)
    dist = [-1] * n
    q: deque[int] = deque([src])
    dist[src] = 0
    while q:
        u = q.popleft()
        for v in adj[u]:
            if dist[v] == -1:
                dist[v] = dist[u] + 1
                q.append(v)
    return dist
```
